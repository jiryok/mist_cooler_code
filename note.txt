
// กำหนดข้อมูลการเชื่อมต่อ Wi-Fi
const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// กำหนด Google App Script Web App URL
const char* googleScriptUrl = "https://script.google.com/macros/s/XXXXXXXXXXXXXXXXXXXXXX/exec";

/**
 * ฟังก์ชันสำหรับเชื่อมต่อ Wi-Fi
 */
void setupWiFi() {
  delay(10);
  Serial.println();
  Serial.print("Connecting to ");
  Serial.println(ssid);

  WiFi.begin(ssid, password);

  // รอจนกว่าจะเชื่อมต่อสำเร็จ
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("");
  Serial.println("WiFi connected");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
}

/**
 * ฟังก์ชันส่งค่าอุณหภูมิและความชื้นไปยัง Google Sheets ผ่าน Google Apps Script
 * @param temp ค่าอุณหภูมิ (องศาเซลเซียส)
 * @param humid ค่าความชื้นสัมพัทธ์ (%)
 */
void recordDataToGoogleSheet(float temp, float humid) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    
    // สร้าง Query Parameter เพื่อส่งไปกับ URL
    String url = String(googleScriptUrl) + "?temp=" + String(temp, 2) + "&humid=" + String(humid, 2);
    
    Serial.println("Sending data to Google Sheets...");
    Serial.print("URL: ");
    Serial.println(url);
    
    // เริ่มต้นเชื่อมต่อ HTTP (เนื่องจาก Google Sheets บังคับใช้ HTTPS)
    // สำหรับการใช้งานทั่วไป สามารถใช้แบบ insecure หรือข้ามการตรวจใบรับรอง SSL ได้ดังนี้
    http.begin(url);
    
    // ตั้งค่าตามระบบของ Google Script ที่มีการ Redirect (302) ไปยังหน้าบริการอื่น
    http.setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS);
    
    // ส่งคำขอแบบ GET
    int httpResponseCode = http.GET();
    
    if (httpResponseCode > 0) {
      String payload = http.getString();
      Serial.print("HTTP Response code: ");
      Serial.println(httpResponseCode);
      Serial.print("Response: ");
      Serial.println(payload);
    } else {
      Serial.print("Error on sending GET: ");
      Serial.println(httpResponseCode);
    }
    
    // ปิดการเชื่อมต่อ
    http.end();
  } else {
    Serial.println("Error: WiFi Disconnected");
  }
}
